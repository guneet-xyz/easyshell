Print one line per matching post inside the loop, and count the lines of the whole loop's output with `wc -l`:

```sh
page=1
while [ "$page" != null ]; do
  response=$(curl -s "http://localhost:4000/posts?page=$page")
  echo "$response" | jq '.posts[] | select(.author == "The Strokes") | .id'
  page=$(echo "$response" | jq .next)
done | wc -l
```
